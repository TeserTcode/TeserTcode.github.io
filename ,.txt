  <script>
    function drawPolygon(ctx, cx, cy, radius, p, q = 1) {
      const points = [];
      for (let i = 0; i < p; i++) {
        const angle = 2 * Math.PI * i / p;
        points.push([
          cx + radius * Math.cos(angle),
          cy + radius * Math.sin(angle)
        ]);
      }

      ctx.beginPath();
      ctx.moveTo(...points[0]);
      let idx = 0;
      const visited = new Set();
      do {
        visited.add(idx);
        idx = (idx + q) % p;
        ctx.lineTo(...points[idx]);
      } while (!visited.has(idx));
      ctx.closePath();
      ctx.stroke();
    }

    function drawCube(ctx, cx, cy, size) {
      const offset = size / 2;
      const front = [
        [cx - offset, cy - offset],
        [cx + offset, cy - offset],
        [cx + offset, cy + offset],
        [cx - offset, cy + offset]
      ];
      const back = front.map(([x, y]) => [x - 40, y - 40]);

      ctx.beginPath();
      front.forEach(([x, y], i) => {
        ctx.moveTo(...front[i]);
        ctx.lineTo(...front[(i + 1) % 4]);
      });
      back.forEach(([x, y], i) => {
        ctx.moveTo(...back[i]);
        ctx.lineTo(...back[(i + 1) % 4]);
      });
      for (let i = 0; i < 4; i++) {
        ctx.moveTo(...front[i]);
        ctx.lineTo(...back[i]);
      }
      ctx.stroke();
    }

    function drawTetrahedron(ctx, cx, cy, size) {
      const h = Math.sqrt(3) / 2 * size;
      const p1 = [cx, cy - size / Math.sqrt(3)];
      const p2 = [cx - size / 2, cy + h / 3];
      const p3 = [cx + size / 2, cy + h / 3];
      const p4 = [cx, cy];

      const lines = [
        [p1, p2], [p2, p3], [p3, p1], // base triangle
        [p1, p4], [p2, p4], [p3, p4]  // sides
      ];

      ctx.beginPath();
      for (const [[x1, y1], [x2, y2]] of lines) {
        ctx.moveTo(x1, y1);
        ctx.lineTo(x2, y2);
      }
      ctx.stroke();
    }

    function draw() {
      const canvas = document.getElementById("canvas");
      const ctx = canvas.getContext("2d");
      ctx.clearRect(0, 0, canvas.width, canvas.height);

      const p = parseInt(document.getElementById("p").value);
      const q = parseInt(document.getElementById("q").value);
      const type = document.getElementById("type").value;

      if (type === "polygon") {
        drawPolygon(ctx, 250, 250, 100, p, 1);
      } else if (type === "star") {
        drawPolygon(ctx, 250, 250, 100, p, q);
      } else if (type === "cube") {
        drawCube(ctx, 250, 250, 120);
      } else if (type === "tetrahedron") {
        drawTetrahedron(ctx, 250, 250, 150);
      }
    }

    draw(); // initial draw
  </script>
