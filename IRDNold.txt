    function rotateCanvas90() {
  const canvas = document.getElementById('inputCanvas');
  const ctx = canvas.getContext('2d');

  // Create an offscreen canvas to store the original content
  const offCanvas = document.createElement('canvas');
  const offCtx = offCanvas.getContext('2d');

  offCanvas.width = canvas.width;
  offCanvas.height = canvas.height;
  offCtx.drawImage(canvas, 0, 0);

  // Swap width and height for 90° rotation
  const newWidth = offCanvas.height;
  const newHeight = offCanvas.width;

  // Resize main canvas
  canvas.width = newWidth;
  canvas.height = newHeight;

  // Apply rotation and draw the image
  ctx.save();
  ctx.translate(newWidth, 0); // move origin to top-right corner
  ctx.rotate(Math.PI / 2);    // rotate 90 degrees clockwise
  ctx.drawImage(offCanvas, 0, 0);
  ctx.restore();
  
  processSinogram(canvas)
}
    
    
    function handleFileUpload(e) {
      const file = e.target.files[0];

        // Calculate dimensions to maintain aspect ratio
        const aspectRatio = img.width / img.height;
        const maxWidth = 300;
        const maxHeight = 200;
        
        let width = img.width;
        let height = img.height;
        
        if (width > maxWidth) {
          width = maxWidth;
          height = width / aspectRatio;
        }
        if (height > maxHeight) {
          height = maxHeight;
          width = height * aspectRatio;
        }
        
        // Draw the input sinogram
        inputCanvas.width = width;
        inputCanvas.height = height;
        inputCtx.drawImage(img, 0, 0, width, height);
        
        // Process the sinogram
        processSinogram(img);
      };
      img.src = URL.createObjectURL(file);
    }

    function showProjectionFFT(transformType) {
      const numAngles = sinogramData.length;

      fftCanvas.height = numBins;
      const fftImageData = fftCtx.createImageData(numAngles, numBins);
      const fftResult = new Array(numAngles);
      
      for (let angleIdx = 0; angleIdx < numAngles; angleIdx++) {
        fftResult[angleIdx] = new Array(numBins);
        const projection = sinogramData[angleIdx];
        
        // Apply FFT with selected transform type
        const fft = fft1D(projection,transformType) 

        
        // Store FFT result
        for (let i = 0; i < numBins; i++) {
          fftResult[angleIdx][i] = fft[i];
        }
        
        // Find max magnitude for scaling
        let maxMag = 0;
        for (let i = 0; i < numBins; i++) {
          const mag = Math.sqrt(fft[i].re**2 + fft[i].im**2);
          maxMag = Math.max(maxMag, mag);
        }
        
        // Draw FFT magnitudes
        for (let i = 0; i < numBins; i++) {
          const idx = (i * numAngles + angleIdx) * 4;
          const mag = Math.sqrt(fft[i].re**2 + fft[i].im**2) / maxMag;
          const val = Math.log(1 + mag * 100) * 20; // Log scale for visualization
          
          fftImageData.data[idx] = val;
          fftImageData.data[idx+1] = val;
          fftImageData.data[idx+2] = val;
          fftImageData.data[idx+3] = 255;
        }
      }
      
      fftCtx.putImageData(fftImageData, 0, 0);
      return fftResult;
    }
    
    function applyFilter(fftResult, filterType) {
      const numAngles = fftResult.length;
      const numBins = fftResult[0].length;
      const filtered = new Array(numAngles);
      
      // Create filter
      const filter = createFilter(numBins, filterType);
      
      filteredFftCanvas.width = numAngles;
      filteredFftCanvas.height = numBins;
      const filteredFftImageData = filteredFftCtx.createImageData(numAngles, numBins);
      
      for (let angleIdx = 0; angleIdx < numAngles; angleIdx++) {
        filtered[angleIdx] = new Array(numBins);
        
        // Apply filter
        for (let i = 0; i < numBins; i++) {
          filtered[angleIdx][i] = {
            re: fftResult[angleIdx][i].re * filter[i],
            im: fftResult[angleIdx][i].im * filter[i]
          };
        }
        
        // Find max magnitude for scaling
        let maxMag = 0;
        for (let i = 0; i < numBins; i++) {
          const mag = Math.sqrt(filtered[angleIdx][i].re**2 + filtered[angleIdx][i].im**2);
          maxMag = Math.max(maxMag, mag);
        }
        
        // Draw filtered FFT
        for (let i = 0; i < numBins; i++) {
          const idx = (i * numAngles + angleIdx) * 4;
          const mag = Math.sqrt(filtered[angleIdx][i].re**2 + filtered[angleIdx][i].im**2) / maxMag;
          const val = Math.log(1 + mag * 100) * 20; // Log scale
          
          filteredFftImageData.data[idx] = val;
          filteredFftImageData.data[idx+1] = val;
          filteredFftImageData.data[idx+2] = val;
          filteredFftImageData.data[idx+3] = 255;
        }
      }
      
      filteredFftCtx.putImageData(filteredFftImageData, 0, 0);
      return filtered;
    }

    function createFilter(length, type) {
      const filter = new Array(length);
      const center = Math.floor(length / 2);
      
      for (let i = 0; i < length; i++) {
        const freq = Math.abs(i - center) / center;
        
  switch (type) {
          case 'ramp':
            filter[i] = freq;
            break;
          case 'inverse-ramp':
            filter[i] = 1 - freq;
            break;
          case 'shepp-logan':
            filter[i] = freq === 0 ? 0 : freq * (Math.sin(Math.PI * freq) / (Math.PI * freq));
            break;
          case 'cosine':
            filter[i] = freq * Math.cos(Math.PI * freq / 2);
            break;
          case 'hann':
            filter[i] = freq * (0.5 * (1 - Math.cos(2 * Math.PI * freq)));
            break;
          case 'hamming':
            filter[i] = freq * (0.54 - 0.46 * Math.cos(2 * Math.PI * freq));
            break;
          case 'blackman':
            filter[i] = freq * (0.42 - 0.5 * Math.cos(2 * Math.PI * freq) + 0.08 * Math.cos(4 * Math.PI * freq));
            break;
          case 'gaussian':
            const sigma = 0.4;
            filter[i] = freq * Math.exp(-0.5 * Math.pow(freq / sigma, 2));
            break;
          case 'lanczos':
            const a = 3;
            filter[i] = (freq === 0) ? 1 : freq * (Math.sin(Math.PI * freq) * Math.sin(Math.PI * freq / a)) / ((Math.PI * freq) ** 2 / a);
            break;
          case 'notch':
            filter[i] = (freq > 0.4 && freq < 0.6) ? 0 : 1;
            break;
          case 'wavelet-like':
            filter[i] = Math.exp(-freq * 8) * Math.sin(6 * Math.PI * freq);
            break;
          case 'explosion':
            filter[i] = Math.min(1, freq ** 3 * 10);
            break;
          case 'rainbow':
            filter[i] = Math.abs(Math.sin(10 * Math.PI * freq));
            break;
          case 'ice-glass':
            filter[i] = Math.pow(Math.sin(Math.PI * freq), 2);
            break;
          case 'mel':
            const melFreq = 2595 * Math.log10(1 + freq * 700);
            filter[i] = melFreq / 4000;
            break;
          case 'spiral-pass':
            filter[i] = (Math.sin(12 * Math.PI * freq) + 1) / 2 * freq;
            break;
          case 'none':
          default:
            filter[i] = 1;
        }
      }
      
      return filter;
    }
    
    function backproject(filteredProjections, size, transformType) {
      const numAngles = filteredProjections.length;
      const numBins = filteredProjections[0].length;
      const recon = new Array(size * size).fill(0);
      
      const center = size / 2;
      const scale = size / numBins;
      
      for (let angleIdx = 0; angleIdx < numAngles; angleIdx++) {
        const angle = (angleIdx / numAngles) * Math.PI;
        const cosAngle = Math.cos(angle);
        const sinAngle = Math.sin(angle);
        
        for (let r = 0; r < numBins; r++) {
          const radius = (r - numBins/2) * scale;
          const value = filteredProjections[angleIdx][r];
          
          // Backproject this value
          for (let t = -center; t <= center; t++) {
            const x = Math.round(center + t * cosAngle - radius * sinAngle);
            const y = Math.round(center + t * sinAngle + radius * cosAngle);
            
            if (x >= 0 && x < size && y >= 0 && y < size) {
              recon[y * size + x] += value;
            }
          }
        }
      }
      
      return recon;
    }
    
    function drawReconstruction(recon) {
      const size = reconstructedCanvas.width;
      const imageData = reconstructedCtx.createImageData(size, size);
      
      // Find max value for normalization
      let maxVal = 0;
      for (let i = 0; i < recon.length; i++) {
        maxVal = Math.max(maxVal, recon[i]);
      }
      
      // Apply brightness and contrast
      const brightnessFactor = brightness;
      const contrastFactor = contrast;
      
      // Draw reconstruction with brightness/contrast adjustments
      for (let y = 0; y < size; y++) {
        for (let x = 0; x < size; x++) {
          const idx = (y * size + x) * 4;
          let val = (recon[y * size + x] / maxVal) * 255;
          
          // Apply contrast
          val = ((val - 127.5) * contrastFactor) + 127.5;
          
          // Apply brightness
          val = val * brightnessFactor;
          
          // Clamp to 0-255
          val = Math.max(0, Math.min(255, val));
          
          imageData.data[idx] = val;
          imageData.data[idx+1] = val;
          imageData.data[idx+2] = val;
          imageData.data[idx+3] = 255;
        }
      }
      
      reconstructedCtx.putImageData(imageData, 0, 0);
    }

      const cy = targetCanvas.height / 2;
      const maxRadius = Math.min(cx, cy);
      
      for (let y = 0; y < h; y++) {
        const angle = (y / h) * Math.PI * 1;
        
        for (let x = 0; x < w; x++) {
          // Map x coordinate to radius (0 to maxRadius)
          const radius = maxRadius/2  - (x / w) * maxRadius;
          
          const px = Math.round(cx + radius * Math.cos(angle));
          const py = Math.round(cy + radius * Math.sin(angle));
          
          const srcIndex = (y + w * x) * 4;
          const destIndex = (py * targetCanvas.width + px) * 4;
          
          if (destIndex >= 0 && destIndex < outData.length - 3) {
            outData[destIndex] = srcData[srcIndex];
            outData[destIndex + 1] = srcData[srcIndex + 1];
            outData[destIndex + 2] = srcData[srcIndex + 2];
            outData[destIndex + 3] = 255;
          }
        }
      }
      
      ctx.putImageData(output, 0, 0);
    }

function sawtoothWave(x) {
  // convert x (radians) to cycles by dividing by 2π and mod 1 to keep in [0,1)
  const cycle = ((x / (2 * Math.PI)) % 1 + 1) % 1; // safe mod for negative x
  return 2 * (cycle - Math.floor(cycle + 0.5));
}

function fft1D(signal, t) {
  const N = signal.length;
  const output = new Array(N);

  for (let k = 0; k < N; k++) {
    let re = 0;
    let im = 0;

    for (let n = 0; n < N; n++) {
      const angle = -2 * Math.PI * k * n / N;
      const { cos, sin } = getBasis(angle, t);
      re += signal[n] * cos;
      im += signal[n] * sin;
    }

    output[k] = { re: re / N, im: im / N };
  }

  return output;
}

function ifft1D(freq, t) {
  const N = freq.length;
  const output = new Array(N);

  for (let n = 0; n < N; n++) {
    let re = 0;
    let im = 0;

    for (let k = 0; k < N; k++) {
      const angle = 2 * Math.PI * k * n / N;
      const { cos, sin } = getBasis(angle, t);
      re += freq[k].re * cos - freq[k].im * sin;
      im += freq[k].re * sin + freq[k].im * cos;
    }

    output[n] = { re: re, im: im };
  }

  return output;
}
