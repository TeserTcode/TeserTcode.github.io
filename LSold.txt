function loadPreset() {
    const presetName = document.getElementById("presetSelect").value;
    if (!presetName) return;

    const preset = presets[presetName];
    document.getElementById("angle").value = preset.angle;
    document.getElementById("main").value = preset.main;

    rules = {};
    moveBehavior = {};

    for (let key in preset.rules) {
        let value = preset.rules[key];

        if (value.startsWith("0")) {
            value = value.slice(1); // Remove leading zero
            moveBehavior[key] = false;
        } else {
            moveBehavior[key] = true;
        }

        rules[key] = value;
    }

    updateRuleUI();
}

        function addRule() {
            const letter = prompt("Enter a new rule letter (A-Z):").toUpperCase();
            if (!letter.match(/^[A-Z]$/) || rules[letter]) return;
            rules[letter] = letter;
            moveBehavior[letter] = true;
            updateRuleUI();
        }

        function updateRuleBindings() {
            document.querySelectorAll(".rule-input").forEach(input => {
                input.oninput = () => rules[input.dataset.letter] = input.value;
            });
            document.querySelectorAll(".move-checkbox").forEach(checkbox => {
                checkbox.onchange = () => moveBehavior[checkbox.dataset.letter] = checkbox.checked;
            });
        }

        function generateLSystem() {
            const iterations = parseInt(document.getElementById('iterations').value);
            let result = document.getElementById('main').value;

            for (let i = 0; i < iterations; i++) {
                result = result.split("").map(ch => rules[ch] || ch).join("");
            }

            drawLSystem(result);
            return result;
        }

   for (let i = 0; i < instructions.length; i++) {

        if (cmd === '+') direction += angle;

        else if (cmd === 'f' || (rules[cmd] && moveBehavior[cmd])) {
            x += moveLength * Math.cos(direction);
            y += moveLength * Math.sin(direction);

        function calculateLacunarity() {
    const canvas = document.getElementById('canvas');
    const ctx = canvas.getContext('2d');
    const boxSize = parseInt(document.getElementById('boxSize').value);
    const boxSpacing = parseInt(document.getElementById('boxSpacing').value);

    let boxCounts = []; // Store counts of filled pixels for each box
    let boxSizes = []; // Store box sizes for regression
    let boxCountsLog = []; // Log of box counts
    let boxSizesLog = []; // Log of box sizes

    // Iterate over the canvas in steps of boxSpacing
    for (let y = 0; y < canvas.height; y += boxSpacing) {
        for (let x = 0; x < canvas.width; x += boxSpacing) {
            // Get the image data for the current box
            const imageData = ctx.getImageData(x, y, boxSize, boxSize).data;
            let filledPixels = 0;

            // Iterate over each pixel in the box
            for (let i = 0; i < imageData.length; i += 4) {
                const r = imageData[i];
                const g = imageData[i + 1];
                const b = imageData[i + 2];
                const a = imageData[i + 3];

                // Check if the pixel is not white (assuming black is the drawn color)
                if ( r!==255  && g!==255 && b!==255 ) {
                    filledPixels++;
                }
            }

            boxCounts.push(filledPixels);
        }
    }

    // Calculate lacunarity
    const mean = boxCounts.reduce((sum, count) => sum + count, 0) / boxCounts.length;
    const variance = boxCounts.reduce((sum, count) => sum + Math.pow(count - mean, 2), 0) / boxCounts.length;

		function linearRegression(x, y) {
			const n = x.length;
    const xMean = x.reduce((sum, val) => sum + val, 0) / n;
    const yMean = y.reduce((sum, val) => sum + val, 0) / n;

    const numerator = x.reduce((sum, val, i) => sum + (val - xMean) * (y[i] - yMean), 0);
    const denominator = x.reduce((sum, val) => sum + Math.pow(val - xMean, 2), 0);

    const slope = numerator / denominator;
    const intercept = yMean - slope * xMean;

    return { slope, intercept };
}

		function calculateHurstExponent() {
    const canvas = document.getElementById('canvas');
    const ctx = canvas.getContext('2d');
    const boxSize = parseInt(document.getElementById('boxSize').value);
    const boxSpacing = parseInt(document.getElementById('boxSpacing').value);

    let data = []; // Store pixel values

    // Iterate over the canvas to collect pixel values
    for (let y = 0; y < canvas.height; y += boxSpacing) {
        for (let x = 0; x < canvas.width; x += boxSpacing) {
            const imageData = ctx.getImageData(x, y, 1, 1).data;
            const r = imageData[0];
            const g = imageData[1];
            const b = imageData[2];
            const a = imageData[3];

            // Check if the pixel is not white
            if (r !== 255 || g !== 255 || b !== 255 || a !== 255) {
                data.push(1); // Filled pixel
            } else {
                data.push(0); // Empty pixel
            }
        }
    }

    // Calculate rescaled range
    const n = data.length;
    const mean = data.reduce((sum, val) => sum + val, 0) / n;
    let cumulativeDeviation = 0;
    let range = 0;
    let stdDev = 0;

    for (let i = 0; i < n; i++) {
        cumulativeDeviation += data[i] - mean;
        if (cumulativeDeviation > range) range = cumulativeDeviation;
        if (cumulativeDeviation < -range) range = -cumulativeDeviation;
        stdDev += Math.pow(data[i] - mean, 2);
    }

    stdDev = Math.sqrt(stdDev / n);

    // Calculate Hurst exponent
    const hurstExponent = Math.log(range / stdDev) / Math.log(n);
    alert(`Hurst Exponent: ${hurstExponent}`);
}
		
		
		function calculateCorrelationDimension() {
    const canvas = document.getElementById('canvas');
    const ctx = canvas.getContext('2d');
    const boxSize = parseInt(document.getElementById('boxSize').value);
    const boxSpacing = parseInt(document.getElementById('boxSpacing').value);

    let points = []; // Store coordinates of non-white pixels

    // Iterate over the canvas to find non-white pixels
    for (let y = 0; y < canvas.height; y += boxSpacing) {
        for (let x = 0; x < canvas.width; x += boxSpacing) {
            const imageData = ctx.getImageData(x, y, 1, 1).data;
            const r = imageData[0];
            const g = imageData[1];
            const b = imageData[2];
            const a = imageData[3];

            // Check if the pixel is not white
            if (r !== 255 || g !== 255 || b !== 255 || a !== 255) {
                points.push({ x, y });
            }
        }
    }

    // Calculate correlation sum
    let correlationSum = 0;
    const r = boxSize/canvas.height; // Distance threshold
    for (let i = 0; i < points.length; i++) {
        for (let j = i + 1; j < points.length; j++) {
            const dx = points[i].x - points[j].x;
            const dy = points[i].y - points[j].y;
            const distance = Math.sqrt(dx * dx + dy * dy);
            if (distance < r) {
                correlationSum++;
            }
        }
    }

    // Calculate correlation dimension
    const correlationDimension = Math.log(correlationSum) / Math.log(r);
    alert(`Correlation Dimension: ${correlationDimension}`);
}
		
		function calculateInformationDimension() {
    const canvas = document.getElementById('canvas');
    const ctx = canvas.getContext('2d');
    const boxSize = parseInt(document.getElementById('boxSize').value);
    const boxSpacing = parseInt(document.getElementById('boxSpacing').value);

    let boxCounts = []; // Store counts of filled pixels for each box
    let totalPixels = 0;

    // Iterate over the canvas in steps of boxSpacing
    for (let y = 0; y < canvas.height; y += boxSpacing) {
        for (let x = 0; x < canvas.width; x += boxSpacing) {
            // Get the image data for the current box
            const imageData = ctx.getImageData(x, y, boxSize, boxSize).data;
            let filledPixels = 0;

            // Iterate over each pixel in the box
            for (let i = 0; i < imageData.length; i += 4) {
                const r = imageData[i];
                const g = imageData[i + 1];
                const b = imageData[i + 2];
                const a = imageData[i + 3];

                // Check if the pixel is not white (assuming black is the drawn color)
                if (r !== 255 || g !== 255 || b !== 255 || a !== 255) {
                    filledPixels++;
                }
            }

            boxCounts.push(filledPixels);
            totalPixels += filledPixels;
        }
    }

    // Calculate Shannon entropy
    let entropy = 0;
    for (const count of boxCounts) {
        if (count > 0) {
            const probability = count / totalPixels;
            entropy -= probability * Math.log(probability);
        }
    }

    // Calculate information dimension
    const informationDimension = entropy / Math.log(canvas.height / boxSize);
    alert(`Information Dimension: ${informationDimension}`);
}
