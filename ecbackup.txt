        function generateCalendar() {
            displayCalendar(currentMonth, currentYear, currentDay);
        }

        function displayCalendar(month, year, highlightDay = null) {
                   const countryFilter = document.getElementById('country-select').value;
            const calendarBody = document.getElementById('calendar-body');
            calendarBody.innerHTML = '';
            const monthYearHeader = document.getElementById('month-year');
            const daysInMonth = new Date(year, month, 0).getDate();
            const firstDay = new Date(year, month - 1, 1).getDay();
            const offset = (firstDay === 0) ? 6 : firstDay - 1;
            const prevDays = new Date(year, month - 1, 0).getDate();

 monthYearHeader.textContent = `${new Intl.DateTimeFormat('en-US', { month: 'long' }).format(new Date(year, month - 1))} - ${year}`;
  let row = document.createElement('tr');
            let nextMonthDay = 1;
 const currentYear = new Date().getFullYear();
            for (let i = 0; i < 42; i++) {
                let cell = document.createElement('td');

                if (i < offset) {
                    cell.classList.add('prev-month');
                    cell.innerHTML = `<span>${prevDays - offset + i + 1}</span>`;
                } else if (i >= offset && i < daysInMonth + offset) {
                    const date = i - offset + 1;
                    cell.textContent = date;
					
                    const tagKey = `${date}-${month}`;
                    if (tags[tagKey] && (countryFilter === 'all' || tags[tagKey].country === countryFilter)) {
                        const tagData = tags[tagKey];
                        cell.setAttribute('data-tag', tagData.name);
                        cell.style.setProperty('--tag-color', tagData.color);
 cell.classList.add(`country-${tagData.country.toLowerCase().replace(/\s+/g, '-')}`);

                        const tagInfo = document.createElement('div');
                        tagInfo.classList.add('tag-info');
                        tagInfo.innerHTML = `<strong>${tagData.name}</strong> (${tagData.country}):<br>${tagData.info}`;
                        cell.appendChild(tagInfo);
                    }

                    if (highlightDay === date) {
                        cell.classList.add('selected-day');
                    }
                } else {
                    cell.classList.add('next-month');
                    cell.innerHTML = `<span>${nextMonthDay++}</span>`;
                }

                row.appendChild(cell);

                if (row.children.length === 7) {
                    calendarBody.appendChild(row);
                    row = document.createElement('tr');
                }
            }
        }

        function changeMonth(delta) {
            currentMonth += delta;
            if (currentMonth < 1) {
                currentMonth = 12;
                currentYear--;
            } else if (currentMonth > 12) {
                currentMonth = 1;
                currentYear++;
            }
            generateCalendar();
        }

        function changeYear(delta) {
            currentYear += delta;
            generateCalendar();
        }

       function setCurrentTime() {
    const now = new Date();
    currentDay = now.getDate();
    currentMonth = now.getMonth() + 1;
    currentYear = now.getFullYear();
    
    // Update input fields with the current date
    document.getElementById('day-input').value = currentDay;
    document.getElementById('month-input').value = currentMonth;
    document.getElementById('year-input').value = currentYear;
    
    generateCalendar();
}

        function changeDay(delta) {
            currentDay += delta;
            const daysInMonth = new Date(currentYear, currentMonth, 0).getDate();
            if (currentDay > daysInMonth) {
                currentDay = 1;
                changeMonth(1);
            } else if (currentDay < 1) {
                changeMonth(-1);
                currentDay = new Date(currentYear, currentMonth, 0).getDate();
            }
            generateCalendar();
        }
